# Testes de tests/. Ligados com cmake -S . -B build -DBUILD_TESTS=ON
# Cada teste e um executavel proprio novo_teste(nome fontes...) compila tests/nome.cpp
# mais as fontes do jogo que ele usa.
function(novo_teste nome)
  add_executable(${nome} ${nome}.cpp ${ARGN})
  target_include_directories(${nome} SYSTEM PRIVATE ${CMAKE_SOURCE_DIR}/external)
  target_include_directories(${nome} PRIVATE ${CMAKE_SOURCE_DIR}/src ${CMAKE_CURRENT_SOURCE_DIR})
  target_compile_definitions(${nome} PRIVATE GAME_ROOT="${CMAKE_SOURCE_DIR}")
  target_link_libraries(${nome} PRIVATE glad glfw Freetype::Freetype)
  if(MSVC)
    target_compile_options(${nome} PRIVATE /utf-8 /W3)
  else()
    target_compile_options(${nome} PRIVATE -Wall -Wextra)
  endif()
  add_test(NAME ${nome} COMMAND ${nome})
endfunction()
